const photoInput = document.getElementById("photo");
const preview = document.getElementById("photoPreview");

if (photoInput && preview) {
    photoInput.addEventListener("change", function () {
        preview.innerHTML = "";
        const file = this.files[0];
        if (!file) return;

        const allowed = ["image/jpeg", "image/png", "image/webp"];
        const maxSize = 5 * 1024 * 1024;

        if (!allowed.includes(file.type)) {
            alert("Only JPG, PNG, and WEBP images are allowed.");
            this.value = "";
            return;
        }

        if (file.size > maxSize) {
            alert("Photo must be 5 MB or smaller.");
            this.value = "";
            return;
        }

        const img = document.createElement("img");
        img.src = URL.createObjectURL(file);
        img.onload = () => URL.revokeObjectURL(img.src);
        preview.appendChild(img);
    });
}

const deleteModal = document.getElementById("deleteModal");
const deleteId = document.getElementById("deleteRequestId");
const deleteLabel = document.getElementById("deleteRequestLabel");
const closeDeleteModal = document.getElementById("closeDeleteModal");
const cancelDelete = document.getElementById("cancelDelete");

document.querySelectorAll("[data-confirm-delete]").forEach(button => {
    button.addEventListener("click", () => {
        if (!deleteModal) return;
        deleteId.value = button.dataset.confirmDelete || "";
        deleteLabel.textContent = `Request #${button.dataset.confirmDelete} – ${button.dataset.title || "Untitled"}`;
        deleteModal.hidden = false;
    });
});

function hideDeleteModal() { if (deleteModal) deleteModal.hidden = true; }
if (closeDeleteModal) closeDeleteModal.addEventListener("click", hideDeleteModal);
if (cancelDelete) cancelDelete.addEventListener("click", hideDeleteModal);
if (deleteModal) deleteModal.addEventListener("click", e => { if (e.target === deleteModal) hideDeleteModal(); });

CREATE DATABASE IF NOT EXISTS campusfix;
USE campusfix;

CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    role ENUM('requester','admin_staff','maintenance','itso','system_admin') NOT NULL DEFAULT 'requester',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE requests (
    id INT AUTO_INCREMENT PRIMARY KEY,
    requester_id INT NOT NULL,
    title VARCHAR(150) NOT NULL,
    category VARCHAR(100) NOT NULL,
    location VARCHAR(150) NOT NULL,
    description TEXT NOT NULL,
    photo VARCHAR(255) NULL,
    assigned_to INT NULL,
    status ENUM('Pending','Assigned','In Progress','Completed') NOT NULL DEFAULT 'Pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT fk_requester
        FOREIGN KEY (requester_id) REFERENCES users(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_assigned
        FOREIGN KEY (assigned_to) REFERENCES users(id)
        ON DELETE SET NULL
);

<?php

$maxFileSize = 5 * 1024 * 1024; // 5 MB
$allowedMimeTypes = [
    "image/jpeg" => "jpg",
    "image/png"  => "png",
    "image/webp" => "webp"
];

if ($_SERVER["REQUEST_METHOD"] === "POST") {
    $title = trim($_POST["title"] ?? "");
    $category = trim($_POST["category"] ?? "");
    $location = trim($_POST["location"] ?? "");
    $description = trim($_POST["description"] ?? "");
    $photo = null;

    if ($title === "" || $category === "" || $location === "" || $description === "") {
        $error = "Please complete all required fields.";
    }

    if (empty($error) && isset($_FILES["photo"]) && $_FILES["photo"]["error"] !== UPLOAD_ERR_NO_FILE) {
        if ($_FILES["photo"]["error"] !== UPLOAD_ERR_OK) {
            $error = "The photo upload failed. Please try again.";
        } elseif ($_FILES["photo"]["size"] > $maxFileSize) {
            $error = "Photo must be 5 MB or smaller.";
        } else {
            $tmpName = $_FILES["photo"]["tmp_name"];
            $mime = (new finfo(FILEINFO_MIME_TYPE))->file($tmpName);
            $imageInfo = @getimagesize($tmpName);

            if (!isset($allowedMimeTypes[$mime]) || $imageInfo === false) {
                $error = "Only valid JPG, PNG, and WEBP images are allowed.";
            } else {
                $uploadDir = __DIR__ . "/uploads/";
                if (!is_dir($uploadDir) && !mkdir($uploadDir, 0755, true)) {
                    $error = "Unable to prepare the upload folder.";
                } else {
                    $photo = bin2hex(random_bytes(16)) . "." . $allowedMimeTypes[$mime];
                    if (!move_uploaded_file($tmpName, $uploadDir . $photo)) {
                        $error = "Unable to save the uploaded photo.";
                        $photo = null;
                    }
                }
            }
        }
    }

    if (empty($error)) {
        $stmt = $conn->prepare("INSERT INTO requests (requester_id, title, category, location, description, photo, status) VALUES (?, ?, ?, ?, ?, ?, 'Pending')");
        $stmt->bind_param("isssss", $_SESSION["user_id"], $title, $category, $location, $description, $photo);

        if ($stmt->execute()) {
            header("Location: my_requests.php?success=" . urlencode("Request submitted successfully."));
            exit;
        } else {
            if ($photo && file_exists(__DIR__ . "/uploads/" . $photo)) {
                unlink(__DIR__ . "/uploads/" . $photo);
            }
            $error = "Unable to submit request.";
        }
    }
}

$allowedStatuses = ["Pending", "Assigned", "In Progress", "Completed"];
if (!in_array($status, $allowedStatuses, true)) {
    header("Location: dashboard.php?error=" . urlencode("Invalid request status."));
    exit;
}

if (in_array($role, ["admin_staff", "system_admin"], true)) {

    if ($assignedTo === "") {
        $stmt = $conn->prepare("UPDATE requests SET assigned_to = NULL, status = ? WHERE id = ?");
        $stmt->bind_param("si", $status, $requestId);
    } else {
        $assignedId = (int)$assignedTo;
        $check = $conn->prepare("SELECT id FROM users WHERE id = ? AND role IN ('maintenance', 'itso')");
        $check->bind_param("i", $assignedId);
        $check->execute();
        if (!$check->get_result()->fetch_assoc()) {
            header("Location: manage_requests.php?error=" . urlencode("Invalid personnel assignment."));
            exit;
        }

        if ($status === "Pending") $status = "Assigned";
        $stmt = $conn->prepare("UPDATE requests SET assigned_to = ?, status = ? WHERE id = ?");
        $stmt->bind_param("isi", $assignedId, $status, $requestId);
    }

    $stmt->execute();
    header("Location: " . $returnTo . (str_contains($returnTo, "?") ? "&" : "?") . "success=" . urlencode("Request updated successfully."));
    exit;
}

if (in_array($role, ["maintenance", "itso"], true)) {

    $stmt = $conn->prepare("UPDATE requests SET status = ? WHERE id = ? AND assigned_to = ?");
    $stmt->bind_param("sii", $status, $requestId, $userId);
    $stmt->execute();

    if ($stmt->affected_rows === 0) {
        header("Location: assigned_requests.php?error=" . urlencode("You can only update requests assigned to your account."));
        exit;
    }

    header("Location: assigned_requests.php?success=" . urlencode("Request status updated successfully."));
    exit;
}

header("Location: dashboard.php?error=" . urlencode("You are not authorized to update requests."));
exit;
